      for (int i = 0; i < num.length(); i++) {
         char s = num.charAt(i);
         int con = (int) s - '0';
         arr[i] = con;

      }

      int odd = 0;
      for (int j = 0; j < num.length(); j++) {
         if (j % 2 == 0) {
            even++;
         } else if (j % 2 == 1) {
            odd++;
         }
      }
      if (odd == even) {

        Arrays.sort(arr);
        System.out.println(Arrays.toString(arr));
      }else{
         Arrays.sort(arr);

         int r=num.length()-1;
         while(l<r){

       }
